
Distinct : 

	- can be used with concat to get full distinc names
		
			+ select distinct concat(fname,lname) from books;

					 or 	

			 + SELECT DISTINCT author_fname, author_lname FROM books;

Order By :

	 - Sort the retrive information by an order of <something>
	
	 - options :
	 		* Desc -> bottom to top

			* ASC -> top to bottom
		ex:
			SELECT * FROM books
			ORDER BY author_lname DESC;

				or

			SELECT book_id, author_fname, author_lname, pages
			FROM books ORDER BY 2 desc;
 		
				or				
	
			SELECT book_id, author_fname, author_lname, pages
			FROM books ORDER BY author_lname, author_fname;

Limit : allows us to limit the amount of results we get back

		ex: 
			SELECT * FROM books
			ORDER BY author_lname DESC limit 5;

				or	
			
			SELECT * FROM books
			ORDER BY author_lname DESC limit 2,10;

Like : Better searching
	
	--> wildcard
		- %something% 

		- underscore --> '____'

		- wildcard is use with the where clause

		-'%something' --> end with something

		- / --> scape key
	
			ex: 

				SELECT title, author_fname, author_lname, pages 
				FROM books
				WHERE author_fname LIKE '%da%';
 
				SELECT title, author_fname, author_lname, pages 
				FROM books
				WHERE title LIKE '%:%';
 
				SELECT * FROM books
				WHERE author_fname LIKE '____';
 
				SELECT * FROM books
				WHERE author_fname LIKE '_a_';


		- / --> scape key 
			ex: 
				-- To select books with '%' in their title:
					SELECT * FROM books
					WHERE title LIKE '%\%%';
 
				-- To select books with an underscore '_' in title:
					SELECT * FROM books
					WHERE title LIKE '%\_%';
